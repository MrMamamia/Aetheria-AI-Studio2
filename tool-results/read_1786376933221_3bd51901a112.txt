   140→function parseArr(s: string | string[] | null | undefined): string[] {
   141→  if (Array.isArray(s)) return s.map(String)
   142→  if (!s) return []
   143→  try {
   144→    const v = JSON.parse(s)
   145→    return Array.isArray(v) ? v.map(String) : []
   146→  } catch {
   147→    return []
   148→  }
   149→}
   150→
   151→function joinArr(arr: string[]): string {
   152→  return arr.join(', ')
   153→}
   154→
   155→function splitArr(s: string): string[] {
   156→  return s
   157→    .split(',')
   158→    .map((p) => p.trim())
   159→    .filter(Boolean)
   160→}
   161→
   162→function entryToDraft(e: LoreEntryT): EntryDraft {
   163→  return {
   164→    id: e.id,
   165→    keys: parseArr(e.keys),
   166→    aliases: parseArr(e.aliases),
   167→    content: e.content ?? '',
   168→    comment: e.comment ?? '',
   169→    enabled: e.enabled,
   170→    position: e.position,
   171→    order: e.order,
   172→    depth: e.depth,
   173→    weight: e.weight,
   174→    activation: e.activation,
   175→    caseSensitive: e.caseSensitive,
   176→    wholeWord: e.wholeWord,
   177→  }
   178→}
   179→
   180→function draftToPayload(d: EntryDraft) {
   181→  return {
   182→    keys: d.keys,
   183→    aliases: d.aliases,
   184→    content: d.content,
   185→    comment: d.comment || null,
   186→    enabled: d.enabled,
   187→    position: d.position,
   188→    order: d.order,
   189→    depth: d.depth,
   190→    weight: d.weight,
   191→    activation: d.activation,
   192→    caseSensitive: d.caseSensitive,
   193→    wholeWord: d.wholeWord,
   194→  }
   195→}

   224→export function LorebooksView() {
   225→  // Lorebook list state
   226→  const {
   227→    data: lorebooks,
   228→    loading: booksLoading,
   229→    error: booksError,
   230→    setData: setBooks,
   231→  } = useFetch<LorebookListItemT[]>('/api/lorebooks')
   232→
   233→  const [search, setSearch] = useState('')
   234→  const [selectedBookId, setSelectedBookId] = useState<string | null>(null)
   235→  const [selectedEntryId, setSelectedEntryId] = useState<string | null>(null)
   236→  const [editorMode, setEditorMode] = useState<'simple' | 'advanced'>('simple')
   237→
   238→  // Filtered lorebook list
   239→  const filteredBooks = useMemo(() => {
   240→    if (!lorebooks) return []
   241→    const q = search.trim().toLowerCase()
   242→    if (!q) return lorebooks
   243→    return lorebooks.filter(
   244→      (b) =>
   245→        b.name.toLowerCase().includes(q) ||
   246→        (b.description ?? '').toLowerCase().includes(q),
   247→    )
   248→  }, [lorebooks, search])
   249→
   250→  // Effective selected book id — falls back to first book when the current
   251→  // selection is missing or invalid. Derived during render (no effect) to
   252→  // avoid cascading re-renders.
   253→  const effectiveBookId = useMemo<string | null>(() => {
   254→    if (!lorebooks || lorebooks.length === 0) return null
   255→    if (selectedBookId && lorebooks.some((b) => b.id === selectedBookId)) {
   256→      return selectedBookId
   257→    }
   258→    return lorebooks[0].id
   259→  }, [lorebooks, selectedBookId])
   260→
   261→  // Selected lorebook detail
   262→  const {
   263→    data: bookDetail,
   264→    loading: bookLoading,
   265→    error: bookError,
   266→    setData: setBookDetail,
   267→  } = useFetch<LorebookDetailT>(
   268→    effectiveBookId ? `/api/lorebooks/${effectiveBookId}` : null,
   269→  )
   270→
   271→  // Effective selected entry id — falls back to first entry when invalid.
   272→  // Derived during render so book switches don't require a state-reset effect.
   273→  const effectiveEntryId = useMemo<string | null>(() => {
   274→    if (!bookDetail) return null
   275→    if (!bookDetail.entries.length) return null
   276→    if (selectedEntryId && bookDetail.entries.some((e) => e.id === selectedEntryId)) {
   277→      return selectedEntryId
   278→    }
   279→    return bookDetail.entries[0].id
   280→  }, [bookDetail, selectedEntryId])
   281→
   282→  // Characters for bound-characters multi-select
   283→  const { data: characters } = useFetch<CharacterT[]>('/api/characters')
   284→
   285→  // ---- Lorebook operations ----
   286→
   287→  const createLorebook = useCallback(async () => {
   288→    try {
   289→      const created = await api<LorebookListItemT>('/api/lorebooks', {
   290→        method: 'POST',
   291→        body: JSON.stringify({ name: 'New Lorebook' }),
   292→      })
   293→      setBooks((prev) => [created, ...(prev ?? [])])
   294→      setSelectedBookId(created.id)
   295→      toast.success('Lorebook created')
   296→    } catch (e) {
   297→      toast.error('Failed to create lorebook', { description: (e as Error).message })
   298→    }
   299→  }, [setBooks])
   300→
   301→  const deleteLorebook = useCallback(
   302→    async (id: string) => {
   303→      try {
   304→        await api(`/api/lorebooks/${id}`, { method: 'DELETE' })
   305→        setBooks((prev) => (prev ?? []).filter((b) => b.id !== id))
   306→        if (selectedBookId === id) {
   307→          setSelectedBookId(null)
   308→          setSelectedEntryId(null)
   309→        }
   310→        toast.success('Lorebook deleted')
   311→      } catch (e) {
   312→        toast.error('Failed to delete lorebook', { description: (e as Error).message })
   313→      }
   314→    },
   315→    [setBooks, selectedBookId],
   316→  )
   317→
   318→  const updateLorebook = useCallback(
   319→    async (id: string, patch: Record<string, unknown>) => {
   320→      const updated = await api<LorebookDetailT>(`/api/lorebooks/${id}`, {
   321→        method: 'PUT',
   322→        body: JSON.stringify(patch),
   323→      })
   324→      // Update list summary (name, enabled, boundCharacters, _count)
   325→      setBooks((prev) =>
   326→        (prev ?? []).map((b) =>
   327→          b.id === id
   328→            ? {
   329→                ...b,
   330→                name: updated.name,
   331→                description: updated.description,
   332→                enabled: updated.enabled,
   333→                boundCharacters: updated.boundCharacters,
   334→                tokenBudget: updated.tokenBudget,
   335→                scanDepth: updated.scanDepth,
   336→              }
   337→            : b,
   338→        ),
   339→      )
   340→      // Update detail cache
   341→      setBookDetail((prev) =>
   342→        prev && prev.id === id
   343→          ? { ...prev, ...updated, entries: prev.entries }
   344→          : prev,
   345→      )
   346→      return updated
   347→    },
   348→    [setBooks, setBookDetail],
   349→  )
   350→
   351→  const toggleBookEnabled = useCallback(
   352→    async (book: LorebookListItemT) => {
   353→      // Optimistic
   354→      setBooks((prev) =>
   355→        (prev ?? []).map((b) =>
   356→          b.id === book.id ? { ...b, enabled: !b.enabled } : b,
   357→        ),
   358→      )
   359→      try {
   360→        await updateLorebook(book.id, { enabled: !book.enabled })
   361→      } catch (e) {
   362→        // Rollback
   363→        setBooks((prev) =>
   364→          (prev ?? []).map((b) =>
   365→            b.id === book.id ? { ...b, enabled: book.enabled } : b,
   366→          ),
   367→        )
   368→        toast.error('Failed to toggle lorebook', { description: (e as Error).message })
   369→      }
   370→    },
   371→    [setBooks, updateLorebook],
   372→  )
   373→
   374→  // ---- Entry operations ----
   375→
   376→  const createEntry = useCallback(async () => {
   377→    if (!effectiveBookId) return
   378→    try {
   379→      const created = await api<LoreEntryT>(
   380→        `/api/lorebooks/${effectiveBookId}/entries`,
   381→        {
   382→          method: 'POST',
   383→          body: JSON.stringify({ keys: [], content: '' }),
   384→        },
   385→      )
   386→      // Insert into detail cache at the end (server orders by order asc, createdAt asc)
   387→      setBookDetail((prev) =>
   388→        prev ? { ...prev, entries: [...prev.entries, created] } : prev,
   389→      )
   390→      setBooks((prev) =>
   391→        (prev ?? []).map((b) =>
   392→          b.id === effectiveBookId
   393→            ? { ...b, _count: { entries: b._count.entries + 1 } }
   394→            : b,
   395→        ),
   396→      )
   397→      setSelectedEntryId(created.id)
   398→      toast.success('Entry created')
   399→    } catch (e) {
   400→      toast.error('Failed to create entry', { description: (e as Error).message })
   401→    }
   402→  }, [effectiveBookId, setBookDetail, setBooks])
   403→
   404→  const deleteEntry = useCallback(
   405→    async (entryId: string) => {
   406→      if (!bookDetail) return
   407→      try {
   408→        await api(`/api/loreentries/${entryId}`, { method: 'DELETE' })
   409→        const remaining = bookDetail.entries.filter((e) => e.id !== entryId)
   410→        setBookDetail((prev) =>
   411→          prev ? { ...prev, entries: remaining } : prev,
   412→        )
   413→        setBooks((prev) =>
   414→          (prev ?? []).map((b) =>
   415→            b.id === bookDetail.id
   416→              ? { ...b, _count: { entries: Math.max(0, b._count.entries - 1) } }
   417→              : b,
   418→          ),
   419→        )
   420→        if (effectiveEntryId === entryId) {
   421→          setSelectedEntryId(remaining[0]?.id ?? null)
   422→        }
   423→        toast.success('Entry deleted')
   424→      } catch (e) {
   425→        toast.error('Failed to delete entry', { description: (e as Error).message })
   426→      }
   427→    },
   428→    [bookDetail, effectiveEntryId, setBookDetail, setBooks],
   429→  )
   430→
   431→  const patchEntryInCache = useCallback(
   432→    (entryId: string, patch: Partial<LoreEntryT>) => {
   433→      setBookDetail((prev) =>
   434→        prev
   435→          ? {
   436→              ...prev,
   437→              entries: prev.entries.map((e) =>
   438→                e.id === entryId ? ({ ...e, ...patch } as LoreEntryT) : e,
   439→              ),
   440→            }
   441→          : prev,
   442→      )
   443→    },
   444→    [setBookDetail],
   445→  )
   446→
   447→  // Toggle an entry's enabled flag from the entry row.
   448→  // Patches cache (so the row re-renders) and dispatches a custom event
   449→  // so the entry editor (which holds its own draft) can stay in sync without
   450→  // triggering a redundant save.
   451→  const toggleEntryEnabled = useCallback(
   452→    async (entryId: string, enabled: boolean) => {
   453→      // Optimistic cache update
   454→      patchEntryInCache(entryId, { enabled })
   455→      // Notify the editor to merge into its draft (and mark as synced)
   456→      window.dispatchEvent(
   457→        new CustomEvent('halcyon:entry-patch', {
   458→          detail: { id: entryId, patch: { enabled } },
   459→        }),
   460→      )
   461→      try {
   462→        await api(`/api/loreentries/${entryId}`, {
   463→          method: 'PUT',
   464→          body: JSON.stringify({ enabled }),
   465→        })
   466→      } catch (e) {
   467→        // Rollback
   468→        patchEntryInCache(entryId, { enabled: !enabled })
   469→        window.dispatchEvent(
   470→          new CustomEvent('halcyon:entry-patch', {
   471→            detail: { id: entryId, patch: { enabled: !enabled } },
   472→          }),
   473→        )
   474→        toast.error('Failed to toggle entry', { description: (e as Error).message })
   475→      }
   476→    },
   477→    [patchEntryInCache],
   478→  )
   479→
   480→  const selectedEntry = useMemo(
   481→    () => bookDetail?.entries.find((e) => e.id === effectiveEntryId) ?? null,
   482→    [bookDetail, effectiveEntryId],
   483→  )
   484→
   485→  return (
   486→    <div className="flex h-full w-full overflow-hidden bg-background">
   487→      {/* LEFT — lorebook list */}
   488→      <aside className="flex w-[280px] shrink-0 flex-col border-r bg-card/30">
   489→        <div className="flex items-center justify-between gap-2 px-3 pb-2 pt-3">
   490→          <div className="flex items-center gap-1.5 text-sm font-medium">
   491→            <BookOpen className="h-4 w-4 text-emerald-600 dark:text-emerald-400" />
   492→            Lorebooks
   493→          </div>
   494→          <Button size="sm" variant="default" className="h-7 gap-1 px-2" onClick={createLorebook}>
   495→            <Plus className="h-3.5 w-3.5" />
   496→            New
   497→          </Button>
   498→        </div>
   499→        <div className="px-3 pb-2">
   500→          <div className="relative">
   501→            <Search className="absolute left-2 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground" />
   502→            <Input
   503→              value={search}
   504→              onChange={(e) => setSearch(e.target.value)}
   505→              placeholder="Search lorebooks…"
   506→              className="h-8 pl-7 text-sm"
   507→            />
   508→          </div>
   509→        </div>
   510→        <Separator />
   511→        <ScrollArea className="flex-1">
   512→          <div className="flex flex-col gap-0.5 p-2">
   513→            {booksLoading ? (
   514→              <div className="flex items-center justify-center gap-2 p-6 text-xs text-muted-foreground">
   515→                <Loader2 className="h-3.5 w-3.5 animate-spin" />
   516→                Loading…
   517→              </div>
   518→            ) : booksError ? (
   519→              <div className="p-4 text-xs text-destructive">{booksError}</div>
   520→            ) : filteredBooks.length === 0 ? (
   521→              <EmptyState
   522→                icon={<BookOpen className="h-6 w-6" />}
   523→                title={search ? 'No matches' : 'No lorebooks'}
   524→                description={
   525→                  search
   526→                    ? 'Try a different search.'
   527→                    : 'Create your first lorebook to start collecting world info.'
   528→                }
   529→                action={
   530→                  !search ? (
   531→                    <Button size="sm" variant="outline" className="mt-2" onClick={createLorebook}>
   532→                      <Plus className="h-3.5 w-3.5" />
   533→                      New Lorebook
   534→                    </Button>
   535→                  ) : null
   536→                }
   537→              />
   538→            ) : (
   539→              filteredBooks.map((b) => (
   540→                <LorebookRow
   541→                  key={b.id}
   542→                  book={b}
   543→                  active={b.id === effectiveBookId}
   544→                  onClick={() => setSelectedBookId(b.id)}
   545→                  onToggleEnabled={() => toggleBookEnabled(b)}
   546→                  onDelete={() => deleteLorebook(b.id)}
   547→                />
   548→              ))
   549→            )}
   550→          </div>
   551→        </ScrollArea>
   552→      </aside>
   553→
   554→      {/* MIDDLE — entry list */}
   555→      <aside className="flex w-[360px] shrink-0 flex-col border-r bg-card/20">
   556→        {bookLoading && !bookDetail ? (
   557→          <div className="flex flex-1 items-center justify-center gap-2 text-xs text-muted-foreground">
   558→            <Loader2 className="h-3.5 w-3.5 animate-spin" />
   559→            Loading entries…
   560→          </div>
   561→        ) : bookError ? (
   562→          <div className="p-4 text-xs text-destructive">{bookError}</div>
   563→        ) : !bookDetail ? (
   564→          <div className="flex flex-1 items-center justify-center p-6 text-center">
   565→            <EmptyState
   566→              icon={<BookOpen className="h-6 w-6" />}
   567→              title="No lorebook selected"
   568→              description="Select a lorebook from the left, or create a new one."
   569→            />
   570→          </div>
   571→        ) : (
   572→          <EntryList
   573→            book={bookDetail}
   574→            characters={characters ?? []}
   575→            selectedEntryId={effectiveEntryId}
   576→            onSelectEntry={setSelectedEntryId}
   577→            onNewEntry={createEntry}
   578→            onDeleteEntry={deleteEntry}
   579→            onToggleEntryEnabled={toggleEntryEnabled}
   580→            onBookSettingsUpdate={updateLorebook}
   581→          />
   582→        )}
   583→      </aside>
   584→
   585→      {/* RIGHT — entry editor */}
   586→      <section className="flex min-w-0 flex-1 flex-col bg-background">
   587→        {selectedEntry && bookDetail ? (
   588→          <EntryEditor
   589→            key={selectedEntry.id}
   590→            entry={selectedEntry}
   591→            mode={editorMode}
   592→            onModeChange={setEditorMode}
   593→            onPatchCache={patchEntryInCache}
   594→            onDelete={deleteEntry}
   595→          />
   596→        ) : (
   597→          <div className="flex flex-1 items-center justify-center p-6">
   598→            <EmptyState
   599→              icon={<Sparkles className="h-6 w-6" />}
   600→              title="No entry selected"
   601→              description="Select an entry from the middle column, or create a new one to begin editing."
   602→            />
   603→          </div>
   604→        )}
   605→      </section>
   606→    </div>
   607→  )
   608→}

   614→function LorebookRow({
   615→  book,
   616→  active,
   617→  onClick,
   618→  onToggleEnabled,
   619→  onDelete,
   620→}: {
   621→  book: LorebookListItemT
   622→  active: boolean
   623→  onClick: () => void
   624→  onToggleEnabled: () => void
   625→  onDelete: () => void
   626→}) {
   627→  const boundIds = parseArr(book.boundCharacters)
   628→  const isGlobal = boundIds.length === 0
   629→
   630→  return (
   631→    <div
   632→      role="button"
   633→      tabIndex={0}
   634→      onClick={onClick}
   635→      onKeyDown={(e) => {
   636→        if (e.key === 'Enter' || e.key === ' ') {

  1143→
  1144→function EntryEditor({
  1145→  entry,
  1146→  mode,
  1147→  onModeChange,
  1148→  onPatchCache,
  1149→  onDelete,
  1150→}: {
  1151→  entry: LoreEntryT
  1152→  mode: 'simple' | 'advanced'
  1153→  onModeChange: (m: 'simple' | 'advanced') => void
  1154→  onPatchCache: (id: string, patch: Partial<LoreEntryT>) => void
  1155→  onDelete: (id: string) => void
  1156→}) {
  1157→  const [draft, setDraft] = useState<EntryDraft>(() => entryToDraft(entry))
  1158→  const [saving, setSaving] = useState(false)
  1159→  const [savedAt, setSavedAt] = useState<number | null>(null)
  1160→
  1161→  // Refs for auto-save snapshot tracking
  1162→  const lastSyncedRef = useRef<string>('')
  1163→  const onPatchCacheRef = useRef(onPatchCache)
  1164→  useEffect(() => {
  1165→    onPatchCacheRef.current = onPatchCache
  1166→  }, [onPatchCache])
  1167→
  1168→  // Listen for external patches (e.g. enabled toggle from entry row)
  1169→  useEffect(() => {
  1170→    function handler(e: Event) {
  1171→      const detail = (e as CustomEvent).detail as {
  1172→        id: string
  1173→        patch: Partial<LoreEntryT>
  1174→      }
  1175→      if (detail.id !== entry.id) return
  1176→      setDraft((prev) => {
  1177→        const next = { ...prev, ...detail.patch }
  1178→        lastSyncedRef.current = JSON.stringify(next)
  1179→        return next
  1180→      })
  1181→    }
  1182→    window.addEventListener('halcyon:entry-patch', handler)
  1183→    return () => window.removeEventListener('halcyon:entry-patch', handler)
  1184→  }, [entry.id])
  1185→
  1186→  // Initialize lastSyncedRef on mount / when entry id changes
  1187→  // (resets so the first render doesn't trigger a save).
  1188→  // draft is intentionally read here without being in deps — we only want
  1189→  // this to run on entry switch, not on every keystroke. (exhaustive-deps is
  1190→  // disabled project-wide.)
  1191→  useEffect(() => {
  1192→    lastSyncedRef.current = JSON.stringify(draft)
  1193→  }, [entry.id])
  1194→
  1195→  // Auto-save with debounce (600ms) — only when draft actually changes from last synced
  1196→  useEffect(() => {
  1197→    const snapshot = JSON.stringify(draft)
  1198→    if (snapshot === lastSyncedRef.current) return
  1199→
  1200→    const t = setTimeout(async () => {

  1530→function SaveIndicator({ saving, savedAt }: { saving: boolean; savedAt: number | null }) {
  1531→  const [, force] = useState(0)
  1532→  // Force a re-render every 30s so "saved Xs ago" stays fresh
  1533→  useEffect(() => {
  1534→    const t = setInterval(() => force((n) => n + 1), 30_000)
  1535→    return () => clearInterval(t)
  1536→  }, [])
  1537→
  1538→  if (saving) {
  1539→    return (
  1540→      <span className="flex items-center gap-1 text-[11px] text-muted-foreground">
  1541→        <Loader2 className="h-3 w-3 animate-spin" />
  1542→        Saving…
  1543→      </span>
  1544→    )
  1545→  }
  1546→  if (savedAt) {
  1547→    const secs = Math.round((Date.now() - savedAt) / 1000)
  1548→    const label = secs < 5 ? 'Saved' : secs < 60 ? `Saved ${secs}s ago` : `Saved ${Math.round(secs / 60)}m ago`
  1549→    return (
  1550→      <span className="flex items-center gap-1 text-[11px] text-emerald-600 dark:text-emerald-400">
  1551→        <span className="h-1.5 w-1.5 rounded-full bg-emerald-500" />
  1552→        {label}
  1553→      </span>
  1554→    )
  1555→  }
  1556→  return null
  1557→}
