   103→function parseGenParams(raw: string | undefined | null): GenParams {
   104→  if (!raw) return { ...DEFAULT_GEN_PARAMS }
   105→  try {
   106→    const parsed = JSON.parse(raw)
   107→    if (parsed && typeof parsed === 'object') {
   108→      return { ...DEFAULT_GEN_PARAMS, ...parsed }
   109→    }
   110→  } catch {
   111→    /* ignore */
   112→  }
   113→  return { ...DEFAULT_GEN_PARAMS }
   114→}
   115→
   116→function parsePromptSettings(raw: string | undefined | null): PromptSettings {
   117→  if (!raw) return { ...DEFAULT_PROMPT_SETTINGS }
   118→  try {
   119→    const parsed = JSON.parse(raw)
   120→    if (parsed && typeof parsed === 'object') {
   121→      return { ...DEFAULT_PROMPT_SETTINGS, ...parsed }
   122→    }
   123→  } catch {
   124→    /* ignore */
   125→  }
   126→  return { ...DEFAULT_PROMPT_SETTINGS }
   127→}

   129→function hydrateFormState(p: PresetListItem): FormState {
   130→  return {

   133→    description: p.description ?? '',

   138→    genParams: parseGenParams(p.genParams),
   139→    promptSettings: parsePromptSettings(p.promptSettings),
   140→  }
   141→}

   173→function formatNumber(value: number | undefined, step?: number): string {
   174→  if (value === undefined || value === null || Number.isNaN(value)) return '—'
   175→  if (step !== undefined && step < 1) {
   176→    const decimals = Math.max(0, -Math.floor(Math.log10(step)))
   177→    return value.toFixed(decimals)
   178→  }
   179→  return String(value)
   180→}

   186→export function PresetsView() {
   187→  const {
   188→    data: presets,
   189→    loading,
   190→    error,
   191→    reload: reloadPresets,
   192→    setData: setPresets,
   193→  } = useFetch<PresetListItem[]>('/api/presets')
   194→
   195→  const { data: apiProfiles } = useFetch<ApiProfileListItem[]>('/api/api-profiles')
   196→
   197→  const [selectedId, setSelectedId] = useState<string | null>(null)
   198→  const [pendingDelete, setPendingDelete] = useState<PresetListItem | null>(null)
   199→
   200→  // Derive the effective selection: fall back to default/first preset.
   201→  // Replaces what would have been a setState-in-effect.
   202→  const effectiveSelectedId = useMemo(() => {
   203→    if (!presets || presets.length === 0) return null
   204→    if (selectedId && presets.find((p) => p.id === selectedId)) return selectedId
   205→    const def = presets.find((p) => p.isDefault) ?? presets[0]
   206→    return def.id
   207→  }, [presets, selectedId])
   208→
   209→  const selectedPreset = useMemo(
   210→    () => presets?.find((p) => p.id === effectiveSelectedId) ?? null,
   211→    [presets, effectiveSelectedId],
   212→  )
   213→
   214→  // Listen for seed completion (so the list refreshes after initial seed).
   215→  useEffect(() => {
   216→    const handler = () => reloadPresets()
   217→    window.addEventListener('halcyon:seeded', handler)
   218→    return () => window.removeEventListener('halcyon:seeded', handler)
   219→  }, [reloadPresets])
   220→
   221→  // ---- Actions ----
   222→
   223→  const handleNew = useCallback(async () => {
   224→    try {
   225→      const created = await api<PresetListItem>('/api/presets', {
   226→        method: 'POST',
   227→        body: JSON.stringify({
   228→          name: 'New Preset',
   229→          providerType: 'zai',
   230→          genParams: { ...DEFAULT_GEN_PARAMS },
   231→          promptSettings: { ...DEFAULT_PROMPT_SETTINGS },
   232→        }),
   233→      })
   234→      setPresets((prev) => (prev ? [created, ...prev] : [created]))
   235→      setSelectedId(created.id)
   236→      toast.success('Preset created')
   237→    } catch (e) {
   238→      toast.error(`Create failed: ${(e as Error).message}`)
   239→    }
   240→  }, [setPresets])
   241→
   242→  const handleDuplicate = useCallback(
   243→    async (p: PresetListItem) => {
   244→      try {
   245→        const gp = parseGenParams(p.genParams)
   246→        const ps = parsePromptSettings(p.promptSettings)
   247→        const created = await api<PresetListItem>('/api/presets', {
   248→          method: 'POST',
   249→          body: JSON.stringify({
   250→            name: `${p.name} (Copy)`,
   251→            description: p.description,
   252→            providerType: p.providerType ?? 'zai',
   253→            modelName: p.modelName,
   254→            apiProfileId: p.apiProfileId,
   255→            genParams: gp,
   256→            promptSettings: ps,
   257→            isDefault: false,
   258→          }),
   259→        })
   260→        setPresets((prev) => (prev ? [created, ...prev] : [created]))
   261→        setSelectedId(created.id)
   262→        toast.success('Preset duplicated')
   263→      } catch (e) {
   264→        toast.error(`Duplicate failed: ${(e as Error).message}`)
   265→      }
   266→    },
   267→    [setPresets],
   268→  )
   269→
   270→  const handleDeleteConfirm = useCallback(async () => {
   271→    const target = pendingDelete
   272→    if (!target) return
   273→    const id = target.id
   274→    setPendingDelete(null)
   275→    try {
   276→      await api(`/api/presets/${id}`, { method: 'DELETE' })
   277→      setPresets((prev) => {
   278→        const next = prev?.filter((p) => p.id !== id) ?? []
   279→        if (selectedId === id) {
   280→          setSelectedId(next.length ? next[0].id : null)
   281→        }
   282→        return next
   283→      })
   284→      toast.success('Preset deleted')
   285→    } catch (e) {
   286→      toast.error(`Delete failed: ${(e as Error).message}`)
   287→      reloadPresets()
   288→    }
   289→  }, [pendingDelete, selectedId, setPresets, reloadPresets])
   290→
   291→  const handleSaved = useCallback(
   292→    (updated: PresetListItem, wasDefault: boolean) => {
   293→      setPresets((prev) => {
   294→        if (!prev) return prev
   295→        return prev.map((p) =>
   296→          p.id === updated.id
   297→            ? { ...p, ...updated, apiProfile: p.apiProfile }
   298→            : wasDefault
   299→              ? { ...p, isDefault: false }
   300→              : p,
   301→        )
   302→      })
   303→    },
   304→    [setPresets],
   305→  )
   306→
   307→  // ============================================================
   308→  // Render

   319→            Retry
   320→          </Button>
   321→        </div>
   322→      </div>
   323→    )
   324→  }
   325→

   330→        <div className="flex items-center justify-between gap-2 border-b p-3">
   331→          <div className="flex items-center gap-2">
   332→            <h2 className="text-sm font-semibold">Presets</h2>

   334→              <Badge variant="secondary" className="h-5 px-1.5 text-[10px]">

   336→              </Badge>
   337→            )}
   338→          </div>
   339→          <Button size="sm" className="h-7 gap-1 px-2" onClick={handleNew}>
   340→            <Plus className="size-3.5" />
   341→            New
   342→          </Button>
   343→        </div>
   344→
   345→        <ScrollArea className="flex-1">
   346→          <div className="p-2">
   347→            {loading && !presets ? (
   348→              <div className="space-y-2 p-1">
   349→                {[0, 1, 2, 3].map((i) => (
   350→                  <div
   351→                    key={i}
   352→                    className="h-14 animate-pulse rounded-md bg-muted"
   353→                  />
   354→                ))}
   355→              </div>
   356→            ) : !presets || presets.length === 0 ? (
   357→              <div className="px-3 py-10 text-center">
   358→                <p className="text-xs text-muted-foreground">
   359→                  No presets yet.
   360→                </p>
   361→                <Button
   362→                  size="sm"
   363→                  variant="outline"
   364→                  className="mt-3"
   365→                  onClick={handleNew}
   366→                >
   367→                  <Plus className="size-3.5" />
   368→                  Create one
   369→                </Button>
   370→              </div>
   371→            ) : (
   372→              <div className="space-y-1">
   373→                {presets.map((p) => (
   374→                  <PresetRow
   375→                    key={p.id}
   376→                    preset={p}
   377→                    selected={p.id === effectiveSelectedId}
   378→                    onSelect={() => setSelectedId(p.id)}
   379→                    onDuplicate={() => handleDuplicate(p)}
   380→                    onDelete={() => setPendingDelete(p)}
   381→                  />
   382→                ))}
   383→              </div>
   384→            )}
   385→          </div>
   386→        </ScrollArea>

   388→
   389→      {/* ---------------- Right: editor ---------------- */}

   396→            onSaved={handleSaved}

   400→          <div className="flex h-full items-center justify-center p-8 text-center">
   401→            <div>
   402→              <p className="text-sm text-muted-foreground">

   406→                <Plus className="size-3.5" />

   426→// ============================================================
   427→// PresetEditor — child component, keyed by preset.id so it
   428→// remounts (and re-hydrates form state) on selection change.
   429→// ============================================================
   430→
   431→function PresetEditor({
   432→  preset,
   433→  apiProfiles,
   434→  onSaved,
   435→  onReloadAfterError,
   436→}: {
   437→  preset: PresetListItem
   438→  apiProfiles: ApiProfileListItem[]
   439→  onSaved: (updated: PresetListItem, wasDefault: boolean) => void
   440→  onReloadAfterError: () => void
   441→}) {
   442→  const [formState, setFormState] = useState<FormState>(() =>
   443→    hydrateFormState(preset),
   444→  )
   445→  const [saveStatus, setSaveStatus] = useState<SaveStatus>('idle')
   446→
   447→  const saveTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   448→  const savedToastTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   449→  const firstRender = useRef(true)
   450→
   451→  // ---- Save pipeline ----
   452→
   453→  const doSave = useCallback(
   454→    async (state: FormState) => {
   455→      try {
   456→        const wasDefault = state.isDefault
   457→        const body = {
   458→          name: state.name || 'Untitled Preset',
   459→          description: state.description.trim() ? state.description : null,
   460→          isDefault: state.isDefault,
   461→          providerType: state.providerType,
   462→          modelName: state.modelName.trim() ? state.modelName.trim() : null,
   463→          apiProfileId: state.apiProfileId || null,
   464→          genParams: state.genParams,
   465→          promptSettings: state.promptSettings,
   466→        }
   467→        const updated = await api<PresetListItem>(
   468→          `/api/presets/${state.id}`,
   469→          { method: 'PUT', body: JSON.stringify(body) },
   470→        )
   471→        onSaved(updated, wasDefault)
   472→        setSaveStatus('saved')
   473→        if (savedToastTimer.current) clearTimeout(savedToastTimer.current)
   474→        savedToastTimer.current = setTimeout(() => {
   475→          setSaveStatus((s) => (s === 'saved' ? 'idle' : s))
   476→        }, 1500)
   477→      } catch (e) {
   478→        setSaveStatus('error')
   479→        toast.error(`Save failed: ${(e as Error).message}`)
   480→        onReloadAfterError()
   481→      }
   482→    },
   483→    [onSaved, onReloadAfterError],
   484→  )
   485→
   486→  // Debounced auto-save: fires 600ms after formState changes.
   487→  // Skips the initial mount so we don't save unchanged data.
   488→  // The 'saving' status is set in the patch helpers (event-handler context),
   489→  // NOT in this effect body — calling setState synchronously in an effect
   490→  // triggers cascading renders and is flagged by the linter.
   491→  useEffect(() => {
   492→    if (firstRender.current) {
   493→      firstRender.current = false
   494→      return
   495→    }
   496→    if (saveTimer.current) clearTimeout(saveTimer.current)
   497→    saveTimer.current = setTimeout(() => {
   498→      void doSave(formState)
   499→    }, 600)
   500→    return () => {
   501→      if (saveTimer.current) clearTimeout(saveTimer.current)
   502→    }
   503→  }, [formState, doSave])
   504→
   505→  // Cleanup toast timer on unmount.
   506→  useEffect(() => {
   507→    return () => {
   508→      if (savedToastTimer.current) clearTimeout(savedToastTimer.current)
   509→    }
   510→  }, [])
   511→
   512→  // ---- Patch helpers ----
   513→
   514→  const patch = useCallback(
   515→    (partial: Partial<FormState>) => {
   516→      setFormState((prev) => ({ ...prev, ...partial }))
   517→      setSaveStatus('saving')
   518→    },
   519→    [],
   520→  )
   521→
   522→  const patchGen = useCallback((partial: Partial<GenParams>) => {
   523→    setFormState((prev) => ({
   524→      ...prev,
   525→      genParams: { ...prev.genParams, ...partial },
   526→    }))
   527→    setSaveStatus('saving')
   528→  }, [])
   529→
   530→  const patchPrompt = useCallback((partial: Partial<PromptSettings>) => {
   531→    setFormState((prev) => ({
   532→      ...prev,
   533→      promptSettings: { ...prev.promptSettings, ...partial },
   534→    }))
   535→    setSaveStatus('saving')
   536→  }, [])
   537→
   538→  const handleResetGen = useCallback(() => {
   539→    patchGen({ ...DEFAULT_GEN_PARAMS })
   540→    toast.info('Generation parameters reset to defaults')
   541→  }, [patchGen])
   542→
   543→  const handleResetPrompt = useCallback(() => {
   544→    patchPrompt({ ...DEFAULT_PROMPT_SETTINGS })
   545→    toast.info('Prompt settings reset to defaults')
   546→  }, [patchPrompt])
   547→
   548→  // ---- Derived ----
   549→
   550→  const capabilities: ProviderCapabilities = useMemo(() => {
   551→    return (
   552→      PROVIDERS[formState.providerType]?.capabilities ??
   553→      PROVIDERS.zai.capabilities
   554→    )
   555→  }, [formState.providerType])
   556→
   557→  // ============================================================
   558→  // Render
   559→  // ============================================================
   560→
   561→  return (
   562→    <div className="flex min-w-0 flex-1 flex-col">
   563→      {/* Editor header */}
   564→      <div className="flex items-center justify-between gap-3 border-b px-5 py-3">
   565→        <div className="min-w-0">
   566→          <div className="flex items-center gap-2">
   567→            <h2 className="truncate text-sm font-semibold">
   568→              {formState.name || 'Untitled Preset'}
   569→            </h2>
   570→            {formState.isDefault && (
   571→              <Badge variant="default" className="h-5 gap-1 px-1.5 text-[10px]">
   572→                <Star className="size-2.5 fill-current" />
   573→                Default
   574→              </Badge>
   575→            )}
   576→          </div>
   577→          <p className="truncate text-xs text-muted-foreground">
   578→            {PROVIDERS[formState.providerType]?.label ?? formState.providerType}
   579→            {formState.modelName ? ` · ${formState.modelName}` : ''}
   580→          </p>
   581→        </div>
   582→        <SaveIndicator status={saveStatus} />
   583→      </div>
   584→
   585→      <ScrollArea className="flex-1">
   586→        <div className="mx-auto max-w-3xl space-y-4 p-5">
   587→          {/* ----- General ----- */}
   588→          <Card>
   589→            <CardHeader>
   590→              <CardTitle className="text-sm">General</CardTitle>
   591→              <CardDescription>
   592→                Basic identity and default status for this preset.
   593→              </CardDescription>
   594→            </CardHeader>
   595→            <CardContent className="space-y-4">
   596→              <div className="space-y-1.5">
   597→                <Label htmlFor="preset-name">Name</Label>
   598→                <Input
   599→                  id="preset-name"
   600→                  value={formState.name}
   601→                  onChange={(e) => patch({ name: e.target.value })}
   602→                  placeholder="e.g. Creative, Balanced, Precise"
   603→                />
   604→              </div>
   605→              <div className="space-y-1.5">
   606→                <Label htmlFor="preset-desc">Description</Label>

  1108→function GenControl({
  1109→  field,
  1110→  value,
  1111→  onChange,
  1112→  disabled,
  1113→}: {
  1114→  field: GenFieldConfig
  1115→  value: unknown
  1116→  onChange: (v: GenParams[keyof GenParams]) => void
  1117→  disabled: boolean
  1118→}) {
  1119→  if (field.type === 'slider') {
  1120→    const v =
  1121→      typeof value === 'number' && !Number.isNaN(value)
  1122→        ? value
  1123→        : (field.min ?? 0)
  1124→    return (
  1125→      <Slider
  1126→        value={[v]}
  1127→        min={field.min}
  1128→        max={field.max}
  1129→        step={field.step}
  1130→        disabled={disabled}
  1131→        onValueChange={(arr) => onChange(arr[0])}
  1132→      />
  1133→    )
  1134→  }
  1135→
  1136→  if (field.type === 'number') {
  1137→    // seed — can be undefined (random)
  1138→    const numVal = typeof value === 'number' ? value : NaN
  1139→    return (
  1140→      <div className="flex items-center gap-2">
  1141→        <Input
  1142→          type="number"
  1143→          disabled={disabled}
  1144→          value={Number.isNaN(numVal) ? '' : String(numVal)}
  1145→          onChange={(e) => {
  1146→            const t = e.target.value.trim()
  1147→            if (t === '') onChange(undefined as never)
  1148→            else {
  1149→              const n = Number(t)
  1150→              if (!Number.isNaN(n)) onChange(n as never)
  1151→            }
  1152→          }}
  1153→          placeholder="Random"
  1154→          className="h-8 max-w-[160px]"
  1155→        />
  1156→        {field.key === 'seed' && !Number.isNaN(numVal) && !disabled && (
  1157→          <Button
  1158→            size="sm"
  1159→            variant="ghost"
  1160→            className="h-8 px-2 text-xs"
  1161→            onClick={() => onChange(undefined as never)}
  1162→          >
  1163→            Clear
  1164→          </Button>
  1165→        )}
  1166→      </div>
  1167→    )
  1168→  }
  1169→
  1170→  if (field.type === 'stringArray') {
  1171→    const arr = Array.isArray(value) ? (value as string[]) : []
  1172→    return (
  1173→      <Input
  1174→        disabled={disabled}
  1175→        value={arr.join(', ')}
  1176→        onChange={(e) => {
  1177→          const next = e.target.value
  1178→            .split(',')
  1179→            .map((s) => s.trim())
  1180→            .filter(Boolean)
  1181→          onChange(next as never)
  1182→        }}
  1183→        placeholder="stop1, stop2, stop3"
  1184→        className="h-8"
  1185→      />
  1186→    )
  1187→  }
  1188→
  1189→  if (field.type === 'boolean') {
  1190→    const checked = Boolean(value)
  1191→    return (
  1192→      <div className="flex items-center gap-2">
  1193→        <Switch
  1194→          checked={checked}
  1195→          disabled={disabled}
  1196→          onCheckedChange={(v) => onChange(v as never)}
  1197→        />
  1198→        <span className="text-xs text-muted-foreground">
  1199→          {checked ? 'Enabled' : 'Disabled'}
  1200→        </span>
  1201→      </div>
  1202→    )
  1203→  }
  1204→
  1205→  return null
  1206→}
